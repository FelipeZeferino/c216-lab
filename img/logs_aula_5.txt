# Listar alunos
GET /api/v1/alunos/
→ [{"id":"GES1","nome":"Maria","email":"maria@inatel.br","curso":"GES","matricula":1}]

# Buscar por ID
GET /api/v1/alunos/GES1

PATCH /api/v1/alunos/GES1
→ {"id":"GES1","nome":"Maria Silva","email":"m.silva@inatel.br","curso":"GES","matricula":1}

# Listar após update
GET /api/v1/alunos/
→ [{"id":"GES1","nome":"Maria Silva","email":"m.silva@inatel.br","curso":"GES","matricula":1}]

DELETE /api/v1/alunos/GES1
→ {"mensagem":"Aluno removido com sucesso"}

→ {"detail":"Aluno nao encontrado"}
